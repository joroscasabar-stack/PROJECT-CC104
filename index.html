Activity 1: Singly Linked List

C++ Code

#include <iostream>
using namespace std;

struct Node {
    int data;
    Node* next;
};

Node* head = nullptr;

void insert(int value) {
    Node* newNode = new Node{value, nullptr};

    if (head == nullptr) {
        head = newNode;
        return;
    }

    Node* current = head;
    while (current->next != nullptr) {
        current = current->next;
    }
    current->next = newNode;
}

void display() {
    Node* current = head;
    while (current != nullptr) {
        cout << current->data << " -> ";
        current = current->next;
    }
    cout << "NULL" << endl;
}

bool search(int value) {
    Node* current = head;
    while (current != nullptr) {
        if (current->data == value) {
            return true;
        }
        current = current->next;
    }
    return false;
}

void remove(int value) {
    if (head == nullptr) return;

    if (head->data == value) {
        Node* temp = head;
        head = head->next;
        delete temp;
        return;
    }

    Node* current = head;
    while (current->next != nullptr &&
           current->next->data != value) {
        current = current->next;
    }

    if (current->next != nullptr) {
        Node* temp = current->next;
        current->next = temp->next;
        delete temp;
    }
}

int main() {
    int id;

    cout << "Enter 5 Student IDs:" << endl;
    for (int i = 0; i < 5; i++) {
        cin >> id;
        insert(id);
    }

    cout << "\nLinked List: ";
    display();

    cout << "Enter ID to search: ";
    cin >> id;

    if (search(id)) {
        cout << id << " found in the list." << endl;
    } else {
        cout << id << " not found in the list." << endl;
    }

    cout << "Enter ID to delete: ";
    cin >> id;
    remove(id);

    cout << "Updated Linked List: ";
    display();

    return 0;
}

Activity 2: Doubly Linked List

C++ Code

#include <iostream>
using namespace std;

struct Node {
    int data;
    Node* prev;
    Node* next;
};

Node* head = nullptr;
Node* tail = nullptr;

void insert(int value) {
    Node* newNode = new Node{value, nullptr, nullptr};

    if (head == nullptr) {
        head = tail = newNode;
    } else {
        tail->next = newNode;
        newNode->prev = tail;
        tail = newNode;
    }
}

void displayForward() {
    Node* current = head;
    while (current != nullptr) {
        cout << current->data << " ";
        current = current->next;
    }
    cout << endl;
}

void displayBackward() {
    Node* current = tail;
    while (current != nullptr) {
        cout << current->data << " ";
        current = current->prev;
    }
    cout << endl;
}

bool searchValue(int value) {
    Node* current = head;
    while (current != nullptr) {
        if (current->data == value) return true;
        current = current->next;
    }
    return false;
}

void updateValue(int oldValue, int newValue) {
    Node* current = head;
    while (current != nullptr) {
        if (current->data == oldValue) {
            current->data = newValue;
            return;
        }
        current = current->next;
    }
}

void deleteValue(int value) {
    Node* current = head;

    while (current != nullptr && current->data != value) {
        current = current->next;
    }
    if (current == nullptr) return;

    if (current->prev != nullptr) {
        current->prev->next = current->next;
    } else {
        head = current->next;
    }

    if (current->next != nullptr) {
        current->next->prev = current->prev;
    } else {
        tail = current->prev;
    }

    delete current;
}

int main() {
    int value, oldValue, newValue;

    cout << "Enter 5 numbers:" << endl;
    for (int i = 0; i < 5; i++) {
        cin >> value;
        insert(value);
    }

    cout << "\nForward: ";
    displayForward();
    cout << "Backward: ";
    displayBackward();

    cout << "Enter value to search: ";
    cin >> value;

    if (searchValue(value)) {
        cout << value << " found in the list." << endl;
    } else {
        cout << value << " not found in the list." << endl;
    }

    cout << "Enter value to update: ";
    cin >> oldValue;
    cout << "Enter new value: ";
    cin >> newValue;
    updateValue(oldValue, newValue);

    cout << "Updated list (forward): ";
    displayForward();

    cout << "Enter value to delete: ";
    cin >> value;
    deleteValue(value);

    cout << "Final list (forward): ";
    displayForward();
    cout << "Final list (backward): ";
    displayBackward();

    return 0;
}

Activity 3: Linear, Circular, and Priority Queues

Three independent queue types with enqueue, dequeue, peek, and display operations.

C++ Code

#include <iostream>
using namespace std;

const int MAX = 5;

// INDEPENDENT QUEUES
int linearQ[MAX];
int linearFront = 0;
int linearRear = -1;

int circularQ[MAX];
int circularFront = 0;
int circularRear = -1;
int circularCount = 0;

int priorityQ[MAX];
int priorityFront = 0;
int priorityRear = -1;

// LINEAR QUEUE
void createLinear() {
    linearFront = 0;
    linearRear = -1;

    cout << "\n===== LINEAR QUEUE =====\n";
    cout << "Enter 5 numbers:\n";

    for (int i = 0; i < MAX; i++) {
        cin >> linearQ[++linearRear];
    }

    cout << "5 numbers successfully entered!\n";
}

void displayLinear() {
    if (linearRear < linearFront) {
        cout << "Queue Empty!\n";
        return;
    }

    cout << "Queue: ";
    for (int i = linearFront; i <= linearRear; i++) {
        cout << linearQ[i] << " ";
    }
    cout << endl;
}

void dequeueLinear() {
    if (linearRear < linearFront) {
        cout << "Queue Empty!\n";
        return;
    }

    cout << "Removed: " << linearQ[linearFront] << endl;

    for (int i = linearFront; i < linearRear; i++) {
        linearQ[i] = linearQ[i + 1];
    }

    linearRear--;

    if (linearRear < linearFront) {
        linearFront = 0;
        linearRear = -1;
    }
}

void enqueueLinear(int x) {
    if (linearRear == MAX - 1) {
        cout << "Queue Full!\n";
        return;
    }

    linearQ[++linearRear] = x;
    cout << "Added: " << x << endl;
}

void peekLinear() {
    if (linearRear < linearFront) {
        cout << "Queue Empty!\n";
    } else {
        cout << "Front: " << linearQ[linearFront] << endl;
    }
}

// CIRCULAR QUEUE
void createCircular() {
    circularFront = 0;
    circularRear = -1;
    circularCount = 0;

    cout << "\n===== CIRCULAR QUEUE =====\n";
    cout << "Enter 5 numbers:\n";

    for (int i = 0; i < MAX; i++) {
        int x;
        cin >> x;

        circularRear = (circularRear + 1) % MAX;
        circularQ[circularRear] = x;
        circularCount++;
    }

    cout << "5 numbers successfully entered!\n";
}

void displayCircular() {
    if (circularCount == 0) {
        cout << "Queue Empty!\n";
        return;
    }

    cout << "Queue: ";
    for (int i = 0; i < circularCount; i++) {
        int index = (circularFront + i) % MAX;
        cout << circularQ[index] << " ";
    }
    cout << endl;
}

void dequeueCircular() {
    if (circularCount == 0) {
        cout << "Queue Empty!\n";
        return;
    }

    cout << "Removed: " << circularQ[circularFront] << endl;
    circularFront = (circularFront + 1) % MAX;
    circularCount--;
}

void enqueueCircular(int x) {
    if (circularCount == MAX) {
        cout << "Queue Full!\n";
        return;
    }

    circularRear = (circularRear + 1) % MAX;
    circularQ[circularRear] = x;
    circularCount++;

    cout << "Added: " << x << endl;
}

void peekCircular() {
    if (circularCount == 0) {
        cout << "Queue Empty!\n";
    } else {
        cout << "Front: " << circularQ[circularFront] << endl;
    }
}

// PRIORITY QUEUE
// Larger number = higher priority
void createPriority() {
    priorityFront = 0;
    priorityRear = -1;

    cout << "\n===== PRIORITY QUEUE =====\n";
    cout << "Enter 5 numbers:\n";

    for (int i = 0; i < MAX; i++) {
        int x;
        cin >> x;

        int j = priorityRear;

        while (j >= 0 && priorityQ[j] < x) {
            priorityQ[j + 1] = priorityQ[j];
            j--;
        }

        priorityQ[j + 1] = x;
        priorityRear++;
    }

    cout << "5 numbers successfully entered!\n";
}

void displayPriority() {
    if (priorityRear < priorityFront) {
        cout << "Queue Empty!\n";
        return;
    }

    cout << "Priority Queue: ";
    for (int i = priorityFront; i <= priorityRear; i++) {
        cout << priorityQ[i] << " ";
    }
    cout << endl;
}

void dequeuePriority() {
    if (priorityRear < priorityFront) {
        cout << "Queue Empty!\n";
        return;
    }

    cout << "Removed: " << priorityQ[0] << endl;

    for (int i = 0; i < priorityRear; i++) {
        priorityQ[i] = priorityQ[i + 1];
    }

    priorityRear--;

    if (priorityRear < priorityFront) {
        priorityFront = 0;
        priorityRear = -1;
    }
}

void enqueuePriority(int x) {
    if (priorityRear == MAX - 1) {
        cout << "Queue Full!\n";
        return;
    }

    int j = priorityRear;

    while (j >= 0 && priorityQ[j] < x) {
        priorityQ[j + 1] = priorityQ[j];
        j--;
    }

    priorityQ[j + 1] = x;
    priorityRear++;

    cout << "Added: " << x << endl;
}

void peekPriority() {
    if (priorityRear < priorityFront) {
        cout << "Queue Empty!\n";
    } else {
        cout << "Highest Priority: " << priorityQ[0] << endl;
    }
}

// LINEAR QUEUE MENU
void linearMenu() {
    int choice, num;

    do {
        cout << "\n================================\n";
        cout << "        LINEAR QUEUE\n";
        cout << "================================\n";
        cout << "1. Enqueue\n2. Dequeue\n3. Peek\n4. Display\n5. Back to Queue Selection\n";
        cout << "================================\nChoice: ";
        cin >> choice;

        switch (choice) {
            case 1:
                cout << "Enter number: ";
                cin >> num;
                enqueueLinear(num);
                break;
            case 2: dequeueLinear(); break;
            case 3: peekLinear(); break;
            case 4: displayLinear(); break;
            case 5: cout << "\nReturning to Queue Selection...\n"; break;
            default: cout << "Invalid Choice!\n";
        }
    } while (choice != 5);
}

// MAIN PROGRAM
int main() {
    int type;

    do {
        cout << "\n\n========================================\n";
        cout << "       QUEUE DATA STRUCTURES\n";
        cout << "========================================\n";
        cout << "1. Linear Queue\n2. Circular Queue\n3. Priority Queue\n4. Exit\n";
        cout << "========================================\nChoose Queue Type: ";
        cin >> type;

        if (type == 1) {
            if (linearRear < linearFront) {
                createLinear();
            } else {
                cout << "\n===== LINEAR QUEUE =====\nExisting queue found.\n";
                displayLinear();
            }
            linearMenu();

        } else if (type == 2) {
            if (circularCount == 0) {
                createCircular();

                cout << "\n===== CIRCULAR QUEUE =====\nExisting queue found.\n";

                createPriority();

            }
            priorityMenu();

        } else if (type == 4) {
            cout << "\nProgram Ended.\n";
        } else {
            cout << "\nInvalid Choice!\n";
        }

    } while (type != 4);

    return 0;
}
